cmake_minimum_required(VERSION 3.25)
project(hmmmgc_switch LANGUAGES C CXX ASM)
get_filename_component(HMMMGC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
set(R "${HMMMGC_ROOT}/src/hmmmgc")
set(W "${HMMMGC_ROOT}/deps/wiicompiled/runtime")
set(G "${R}/build/native/translation/game")
string(REPLACE ".section .rdata,\"dr\"" ".section .rodata,\"a\",%progbits" _b "${_b}")
file(GLOB SHARDS CONFIGURE_DEPENDS "${G}/build_shards/base_common/*.cpp" "${G}/build_shards/base_registration/*.cpp" "${G}/build_shards/base_dispatch/*.cpp")
add_library(hmmmgc_game STATIC ${SHARDS} "${W}/src/fpu_helpers.cpp" "${W}/src/ppc_quantized.cpp")
target_include_directories(hmmmgc_game PRIVATE "${W}/include" "${G}" "${G}/build_shards/base_common" "${R}/runtime/src" "${R}/runtime/mod_api")
target_compile_definitions(hmmmgc_game PRIVATE HMMMGC_SWITCH=1)
target_compile_options(hmmmgc_game PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math -ffp-contract=off -O2 -fomit-frame-pointer -fno-tree-slp-vectorize -fno-semantic-interposition -w>)
set(MODS jamie_marriage player_speed carry_speed far_camera marry_anyone longer_days gentle_stamina faster_tools cheats)
set(_o)
set(_d "")
set(_g "")
foreach(m IN LISTS MODS)
  add_library(mod_${m} OBJECT "${R}/mods/${m}/mod.c")
  target_include_directories(mod_${m} PRIVATE "${R}/runtime/mod_api" "${R}/mods/include" "${R}/build")
  target_compile_definitions(mod_${m} PRIVATE moderngekko_get_mod=moderngekko_get_mod_${m})
  target_compile_options(mod_${m} PRIVATE -std=c11 -O2 -w)
  list(APPEND _o $<TARGET_OBJECTS:mod_${m}>)
  string(APPEND _d "extern \"C\" const ModernGekkoModDesc* moderngekko_get_mod_${m}(void);\n")
  string(APPEND _g "    moderngekko_get_mod_${m},\n")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/native_mod_list.h" "#pragma once\n#include \"moderngekko/mod_abi.h\"\n${_d}static const ModernGekkoGetModFn COMPILED_IN_MODS[]={\n${_g}nullptr};\n")
add_library(hmmmgc_mods STATIC ${_o})
